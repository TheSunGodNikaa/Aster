-- ============================================================
-- Pharma wholesale billing software — SQLite schema
-- Redesigned from the legacy VB6 / MS Access (database.MDB) structure.
--
-- Design principles applied:
--   1. Money stored as INTEGER paise (never REAL) — avoids float
--      rounding errors that the old schema had everywhere.
--   2. Real foreign keys instead of denormalized name-copies in
--      every child row (the old schema repeated Customer_Name,
--      Product_Name etc. in every transaction row).
--   3. Batch/expiry is a first-class table, not flattened into
--      every stock and sale row — this is what makes FEFO issuing
--      and expiry alerts a simple query instead of manual work.
--   4. One unified `invoices` + `invoice_items` pair with a
--      `invoice_type` column, instead of ~8 near-duplicate
--      parent/child table pairs (Sales_Parent/Child,
--      Purchase_Parent/Child, Credit_Parent/Child, Debit_Parent/
--      Child, Delivery_Parent/Child, DSales_Parent/Child...).
--   5. One `ledger_entries` table for both customer and supplier
--      money movement, instead of separate Receipt/payment/
--      SalesAccount/PurchaseAccount tables that had to be kept
--      in sync by hand.
--   6. GSTR1 / GSTR3B / HSN summary are NOT stored tables (the old
--      DB stored and re-synced these manually, which risks
--      staleness). Generate them as SQL views or on-demand report
--      queries from invoices/invoice_items instead — see bottom.
-- ============================================================

-- ---------- Products & stock ----------
CREATE TABLE products (
    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    code              TEXT UNIQUE,
    name              TEXT NOT NULL,
    description       TEXT,
    pack              TEXT,             -- e.g. '10 S', '30 GM'

    uqc               TEXT DEFAULT 'NOS',   -- unit quantity code for GST filing, e.g. NOS, KGS, BOT, STRP

-- Each purchased lot of a product. This is the heart of pharma stock control.
CREATE TABLE batches (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    product_id          INTEGER NOT NULL REFERENCES products(id),
    batch_no            TEXT NOT NULL,
    expiry_date         DATE,                 -- store as real DATE, not the old free-text '01/2027'
    mrp_paise           INTEGER NOT NULL DEFAULT 0,
    purchase_price_paise INTEGER NOT NULL DEFAULT 0,
    sales_price_paise   INTEGER NOT NULL DEFAULT 0,
    ptr_paise           INTEGER NOT NULL DEFAULT 0,   -- price to retailer
    qty_on_hand         INTEGER NOT NULL DEFAULT 0,
    free_qty_on_hand    INTEGER NOT NULL DEFAULT 0,
    created_at          TEXT DEFAULT (datetime('now')),
    UNIQUE (product_id, batch_no)
);
CREATE INDEX idx_batches_product ON batches(product_id);
CREATE INDEX idx_batches_expiry ON batches(expiry_date);   -- powers expiry alerts

-- Full audit trail of every stock change (replaces StockAdjustment/StockModify/
-- the implicit stock changes buried inside Purchase_Child/Sales_Child).
CREATE TABLE stock_movements (
    id               INTEGER PRIMARY KEY AUTOINCREMENT,
    batch_id         INTEGER NOT NULL REFERENCES batches(id),
    invoice_item_id  INTEGER REFERENCES invoice_items(id),  -- null for manual adjustments
    movement_type    TEXT NOT NULL CHECK (movement_type IN
                        ('purchase','purchase_return','sale','sale_return',
                         'adjustment','transfer_in','transfer_out')),
    qty_change       INTEGER NOT NULL,     -- +ve = stock in, -ve = stock out
    free_qty_change  INTEGER NOT NULL DEFAULT 0,
    balance_after    INTEGER NOT NULL,     -- qty_on_hand snapshot after this movement
    remarks          TEXT,
    created_at       TEXT DEFAULT (datetime('now'))
);
CREATE INDEX idx_stock_movements_batch ON stock_movements(batch_id);

-- ---------- Invoices (unified transaction table) ----------
-- One table for every document type instead of 8 parent/child pairs.
CREATE TABLE invoices (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    invoice_no          TEXT NOT NULL,
    invoice_type        TEXT NOT NULL CHECK (invoice_type IN
                            ('purchase','purchase_return','sale','sale_return',
                             'delivery_challan','credit_note','debit_note')),
    financial_year      TEXT NOT NULL,        -- e.g. '2026-27', drives numbering + archiving/reports
    invoice_date        DATE NOT NULL DEFAULT (date('now')),
    note_type           TEXT CHECK (note_type IN ('financial','commercial')), -- CNote/DNote 'Type': GST-relevant vs internal adjustment
    note_reason         TEXT,                 -- CNote/DNote 'Reason', e.g. 'Rate difference', 'Damaged goods'
    party_type          TEXT NOT NULL CHECK (party_type IN ('customer','supplier')),
    customer_id         INTEGER REFERENCES customers(id),
    supplier_id         INTEGER REFERENCES suppliers(id),
    party_gstin_snap    TEXT,                  -- GSTIN as it was AT THE TIME of this invoice — legal documents must stay frozen
    agent_id            INTEGER REFERENCES agents(id),
    reference_invoice_id INTEGER REFERENCES invoices(id),  -- e.g. a sale_return points back at the sale
    subtotal_paise      INTEGER NOT NULL DEFAULT 0,
    discount_paise      INTEGER NOT NULL DEFAULT 0,
    taxable_paise       INTEGER NOT NULL DEFAULT 0,
    cgst_paise          INTEGER NOT NULL DEFAULT 0,
    sgst_paise          INTEGER NOT NULL DEFAULT 0,
    igst_paise          INTEGER NOT NULL DEFAULT 0,
    round_off_paise     INTEGER NOT NULL DEFAULT 0,
    other_charges_paise INTEGER NOT NULL DEFAULT 0,
    net_amount_paise    INTEGER NOT NULL DEFAULT 0,
    status              TEXT NOT NULL DEFAULT 'posted' CHECK (status IN ('draft','posted','cancelled')),
    created_by          INTEGER REFERENCES users(id),
    created_at          TEXT DEFAULT (datetime('now')),
    CHECK (
        (party_type = 'customer' AND customer_id IS NOT NULL AND supplier_id IS NULL) OR
        (party_type = 'supplier' AND supplier_id IS NOT NULL AND customer_id IS NULL)
    )
);
CREATE UNIQUE INDEX idx_invoices_no_type_fy ON invoices(invoice_type, financial_year, invoice_no);
CREATE INDEX idx_invoices_date ON invoices(invoice_date);
CREATE INDEX idx_invoices_fy ON invoices(financial_year);
CREATE INDEX idx_invoices_customer ON invoices(customer_id);
CREATE INDEX idx_invoices_supplier ON invoices(supplier_id);

CREATE TABLE invoice_items (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    invoice_id          INTEGER NOT NULL REFERENCES invoices(id) ON DELETE CASCADE,
    sno                 INTEGER NOT NULL,           -- line number on the printed invoice
    product_id          INTEGER NOT NULL REFERENCES products(id),
    batch_id            INTEGER REFERENCES batches(id),
    -- snapshots: what the product looked like AT THE TIME of this invoice.
    -- Products get renamed/repriced later; old invoices must stay frozen.
    product_name_snap   TEXT NOT NULL,
    hsn_snap            TEXT,
    pack_snap           TEXT,
    qty                 INTEGER NOT NULL,
    free_qty            INTEGER NOT NULL DEFAULT 0,
    mrp_paise           INTEGER NOT NULL DEFAULT 0,
    ptr_paise           INTEGER NOT NULL DEFAULT 0,   -- price to retailer, used for margin reporting
    rate_paise          INTEGER NOT NULL DEFAULT 0,   -- price actually billed at
    discount_percent    REAL NOT NULL DEFAULT 0,
    gst_percent         REAL NOT NULL DEFAULT 0,
    cgst_paise          INTEGER NOT NULL DEFAULT 0,
    sgst_paise          INTEGER NOT NULL DEFAULT 0,
    igst_paise          INTEGER NOT NULL DEFAULT 0,
    cess_paise          INTEGER NOT NULL DEFAULT 0,
    line_total_paise    INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX idx_invoice_items_invoice ON invoice_items(invoice_id);
CREATE INDEX idx_invoice_items_batch ON invoice_items(batch_id);

-- ---------- Money ledger (unified receipts + payments) ----------
CREATE TABLE ledger_entries (
    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    party_type        TEXT NOT NULL CHECK (party_type IN ('customer','supplier')),
    customer_id       INTEGER REFERENCES customers(id),
    supplier_id       INTEGER REFERENCES suppliers(id),
    entry_date        DATE NOT NULL DEFAULT (date('now')),
    entry_type        TEXT NOT NULL CHECK (entry_type IN
                        ('invoice','receipt','payment','opening_balance')),
    invoice_id        INTEGER REFERENCES invoices(id),
    debit_paise       INTEGER NOT NULL DEFAULT 0,   -- party owes you more (sale, or you owe supplier more)
    credit_paise      INTEGER NOT NULL DEFAULT 0,   -- money received/paid, reduces balance
    cheque_no         TEXT,
    cheque_date       DATE,
    remarks           TEXT,
    created_at        TEXT DEFAULT (datetime('now'))
);
CREATE INDEX idx_ledger_customer ON ledger_entries(customer_id, entry_date);
CREATE INDEX idx_ledger_supplier ON ledger_entries(supplier_id, entry_date);

-- ---------- Printer profiles (dot-matrix vs laser) ----------
CREATE TABLE printers (

    name         TEXT NOT NULL,
    kind         TEXT NOT NULL CHECK (kind IN ('dot_matrix','laser','thermal')),
    control_codes TEXT,        -- raw ESC/POS-style control sequence, old Printer.Charcter
    is_default   INTEGER NOT NULL DEFAULT 0
);

-- ---------- Invoice numbering ----------
-- Generates sequential numbers per (invoice_type, financial_year) so each
-- document type restarts its own sequence every financial year, mirroring
-- how the old software began a fresh MDB per year. No more separate DB

CREATE TABLE invoice_counters (
    invoice_type    TEXT NOT NULL,
    financial_year  TEXT NOT NULL,
    last_number     INTEGER NOT NULL DEFAULT 0,
    PRIMARY KEY (invoice_type, financial_year)
);

-- ============================================================
-- Triggers: keep batches.qty_on_hand in sync with stock_movements
-- so you never have to remember to update both tables by hand.
-- ============================================================
CREATE TRIGGER trg_stock_movement_apply
AFTER INSERT ON stock_movements
BEGIN
    UPDATE batches
    SET qty_on_hand = qty_on_hand + NEW.qty_change,
        free_qty_on_hand = free_qty_on_hand + NEW.free_qty_change
    WHERE id = NEW.batch_id;
END;

-- ============================================================
-- Reporting views — generated on demand, never stored.
-- This replaces GSTR1 / GSTR1NEW / GSTR3B / HSN_SUMMARY tables,
-- which the old system had to keep manually synced (a source of bugs).
-- ============================================================

-- HSN-wise summary for the current data set
CREATE VIEW v_hsn_summary AS
SELECT
    ii.hsn_snap                                   AS hsn,
    SUM(ii.qty)                                   AS total_qty,
    SUM(ii.line_total_paise)                      AS total_value_paise,
    SUM(ii.line_total_paise - ii.cgst_paise - ii.sgst_paise - ii.igst_paise) AS taxable_paise,
    SUM(ii.cgst_paise)                            AS cgst_paise,
    SUM(ii.sgst_paise)                            AS sgst_paise,
    SUM(ii.igst_paise)                            AS igst_paise
FROM invoice_items ii
JOIN invoices i ON i.id = ii.invoice_id
WHERE i.invoice_type = 'sale' AND i.status = 'posted'
GROUP BY ii.hsn_snap;

-- Batches nearing expiry (e.g. within 90 days) — for daily alerts
CREATE VIEW v_expiring_soon AS
SELECT b.id AS batch_id, p.name AS product_name, b.batch_no, b.expiry_date, b.qty_on_hand
FROM batches b
JOIN products p ON p.id = b.product_id
WHERE b.qty_on_hand > 0
  AND b.expiry_date IS NOT NULL
  AND julianday(b.expiry_date) - julianday('now') <= 90
ORDER BY b.expiry_date ASC;

-- Customer outstanding balance
CREATE VIEW v_customer_balance AS
SELECT customer_id, SUM(debit_paise) - SUM(credit_paise) AS balance_paise
FROM ledger_entries
WHERE party_type = 'customer'
GROUP BY customer_id;

-- Supplier outstanding balance
CREATE VIEW v_supplier_balance AS
SELECT supplier_id, SUM(debit_paise) - SUM(credit_paise) AS balance_paise
FROM ledger_entries
WHERE party_type = 'supplier'
GROUP BY supplier_id;
